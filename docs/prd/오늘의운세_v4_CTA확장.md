## 7. AI 출력 JSON 스키마 — 오늘의 운세 (사주 있음)

```typescript
interface DailyFortuneResult {
  date: string                 // "2026-04-19"
  score: number
  grade: 'S' | 'A' | 'B' | 'C' | 'D'
  headline: string             // 한 줄 요약 (예: "창의성이 빛나는 날")
  summary: string              // 3~4문장, ~200자
  keywords: string[]           // 5개

  sajuTone: string             // 250자, 일간·오행 연결 톤

  ohengFlow: {
    sajuDistribution: Record<OhengKey, number>
    todayEmphasis: OhengKey    // 오늘 강세 오행
    insight: string            // 1~2문장
  }

  scores: {
    재물운: { score: number, desc: string }   // 각 2~3문장 펼침 대응
    애정운: { score: number, desc: string }
    건강운: { score: number, desc: string }
    성장운: { score: number, desc: string }
  }

  timeSlots: Array<{
    slot: '오전' | '오후' | '저녁' | '심야'
    range: string              // "6-12시"
    score: number
    tip: string                // 2줄, ~80자
  }>  // 4개

  luckyDirection: {
    direction: '동' | '서' | '남' | '북' | '동남' | ...
    desc: string               // "해가 뜨는 방향..."
  }

  doList: string[]             // 3개, 각 ~50자
  dontList: string[]           // 3개

  dailyMessage: string         // 150자, 개운 한 마디

  luckyProfile: {
    color: { name, hex, reason }  // 공개
    // 숫자·방위 상세·보충오행은 별도 상품
  }
}
```
